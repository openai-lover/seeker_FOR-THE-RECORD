cmake_minimum_required(VERSION 3.22)
project(reflection_native LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set_property(GLOBAL PROPERTY JOB_POOLS reflection_compile=4)
set(CMAKE_JOB_POOL_COMPILE reflection_compile)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_CPU_KLEIDIAI OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
include(FetchContent)
# The hash prevents a changed tag from silently changing the shipped engine.
set(FETCHCONTENT_BASE_DIR "${CMAKE_CURRENT_BINARY_DIR}/_deps" CACHE PATH "" FORCE)
FetchContent_Declare(llama
    URL https://api.github.com/repos/ggml-org/llama.cpp/zipball/v0.5.0
    URL_HASH SHA256=78debec727e31cd4f85b3795a94c2864db66d2687a81c5c36e9717fec785568c)
FetchContent_MakeAvailable(llama)
add_library(reflection_native SHARED reflection_core.cpp reflection_jni.cpp)
target_link_libraries(reflection_native PRIVATE llama)
target_link_options(reflection_native PRIVATE "-Wl,-z,max-page-size=16384")
